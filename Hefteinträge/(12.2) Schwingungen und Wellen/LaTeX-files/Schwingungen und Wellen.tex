	\section{Die harmonische Schwingung}
	z.B. Feder-Masse-Pendel (s.o.)

	\noindent
	\begin{minipage}{0.55\textwidth}
		\begin{alignat}{1}
			F_G &= g \cdot m\\
			F_T &= m \cdot a =\\
			F_D &= D \cdot x \qquad \text{x: Auslenkung (eig. } \Delta x)\\
		\end{alignat}
		\vspace{-1.5cm}
		\begin{alignat}{1}
			F_G + F_D &= F_T\\
			-g m + D x &= m a\\
			-gm + Dx &= m \ddot{x} \qquad \text{Inhomogene DGl 2. Ordnung}
		\end{alignat}
	\end{minipage}
	\hfill
	\fbox{
		\begin{minipage}[t]{0.3\textwidth}
			\textbf{Einschub}
			\vspace{-0.3cm}
			\begin{alignat}{1}
				x(t) &= \frac{1}{2} a t^2 + v_0 t + s_0\\
				v(t) &= \dot{x}(t) = at + v_0\\
				a(t) &= \dot{v}(t) = \ddot{x}(t) = a
			\end{alignat}
		\end{minipage}
	}
	\vspace{1cm}\\
	\textbf{Ansatz: } $x (t) = x_0 \sin (\omega t + \varphi)$\\
	\vspace{-0.5cm}
	\begin{alignat}{2}
		\omega = 2\pi f &= \frac{2\pi}{T} &\quad&:\text{Winkelgeschwindigkeit}\\
		f      &= \frac{1}{T}              &\quad&:\text{Frequenz}\\
		& T  &\quad&:\text{Periodendauer}\\
		& \varphi  &\quad&:\text{Phase}
	\end{alignat}
	\vspace{-1cm}
	\begin{alignat}{1}
		\dot{x} (t) &= x_0 \omega \cos (\omega t + \varphi)\\
		\ddot{x} (t) &= -x_0 \omega^2 \sin (\omega t + \varphi)\\
	\end{alignat}


	\noindent
	\textbf{Einsetzen} \\
	$-g m + D x_0 \sin (\omega t + \varphi) = -m x_0 \omega^2 \sin (\omega t + \varphi)$
	\vspace{0.2cm}\\
	$\rightarrow$ So geht's nicht weiter, weil wir nicht nach $\omega$ auflösen können $\sqrt{-1}$\\
	$\rightarrow$ Inhomogenität (nicht abhängig von x oder $\ddot{x}$) können wir auch nicht verarbeiten\\
	Inhomogene DGL werden gelöst, indem man die homogene DGL löst und einen Spezialfall für die inhomogene hinzuaddiert.\\
	Homogen mit richtigem Vorzeichen:
	\begin{alignat}{1}
		Dx - m \ddot{x} = 0 \quad \leftarrow \text{homogen}\\
		Dx_0 \sin (\omega t + \varphi) - m x_0 \omega^2 \sin (\omega t + \varphi) &= 0\\
		(D - m \omega^2) x_0 \sin (\omega t + \varphi) &= 0\\
		D - m \omega^2 = 0 \qquad \vee \qquad x_0 \sin (\omega t + \varphi) &= 0 \qquad \text{(Laaangweilig, weil dann nichts schwingt)}\\
	\end{alignat}
	\vspace{-1cm}
	\begin{alignat}{1}
		\Rightarrow D &= m \omega^2\\
		\Rightarrow \omega &= \sqrt{\frac{D}{m}}\\
		\Rightarrow \frac{2 \pi}{T} &= \sqrt{\frac{D}{m}}\\
		\Rightarrow T &= 2 \pi \sqrt{\frac{m}{D}}
	\end{alignat}


	\newpage
	\noindent
	Lösung der homogenen DGL \qquad $x(t) = x_0 \sin \Big( \sqrt{\frac{D}{m}} \cdot t + \varphi \Big)$\\
	Lösung der inhomogenen DGL \qquad $x(t) = x_0 \sin \Big( \sqrt{\frac{D}{m}} t + \varphi \Big)$\\
	\\
	$Dx - mg + m \ddot{x} = 0$\\
	``-mg'' muss beim Einsetzen von x(t) herauskommen\\
	$\Rightarrow x(t) = x_0 \sin \Big( \sqrt{\frac{D}{m}} t + \varphi \Big) + \frac{mg}{D}$\\
	\\
	\textbf{Test durch Einsetzen:}\\
	\begin{alignat}{1}
		\dot{x}&=x_0\sqrt{\tfrac Dm}\cos \Big(\sqrt{\tfrac Dm}t+\varphi \Big)\\
		\ddot{x}&=-x_0\frac Dm \sin \Big( \sqrt{\tfrac Dm}t+\varphi \Big)\\
		Dx-mg+m\ddot{x}&=0\\
		D\Big(x_0 \sin \Bigg( \sqrt{\tfrac Dm}t+\varphi \Bigg)+\frac{mg}D \Big)-mg+m \Big(-x_0\frac D m \sin \Bigg(\sqrt{\frac Dm}t+\varphi\Bigg) \Big)&=0\\
		Dx_0 \sin \Big(\sqrt{\tfrac Dm}t+\varphi \Big)+mg-mg-x_0D \sin \Big(\sqrt{\tfrac Dm}t+\varphi \Big)&=0\\
		0&=0\\
	\end{alignat}
	\vspace{0.5cm}\\
	\subsection{Hausaufgabe:}
	\vspace{-0.3cm}
	\begin{minipage}[htpb]{0.6\textwidth}
		\vspace{0pt}
		\begin{alignat}{3}
			& \hspace{1cm} & F_D &= F_G              &\hspace{1cm}& \mid F_G = -mg\\
			&            & D x &= mg               &      & \\
			\Leftrightarrow &  & x   &= \frac{mg}{D}     && \mid x = -\Delta x\\
			\Rightarrow &      & x   &= -\frac{mg}{D}    &      &
		\end{alignat}
	\end{minipage}
	\hfill
	\begin{minipage}[b]{0.4\textwidth}
		\vspace{0pt}
		\begin{alignat}{1}
			F_D + F_G &= 0\\
			D x - m g &= 0
		\end{alignat}
	\end{minipage}
	\vspace{0.3cm}
	\subsection{Definition: } Eine Schwingung heißt harmonisch, wenn die Rückstellkraft (hier $F_D = Dx$) proportional zur Auslenkung ist. Die dazugehörige DGL wird dann durch eine Sinus-(bzw. Kosinus-)Funktion gelöst.\\
	\vspace{-0.3cm}\\
	Unsere Lösung $x(t) = x_0 \sin \Big( \sqrt{\tfrac{D}{m}} t + \varphi \Big) + \frac{mg}{D}$ muss noch durch die Startbedingungen festgelegt werden\\

	\section{Energiebilanz beim Feder-Masse-Pendel}
	(Auslenkung) $x_A = x_0 \sin (\omega t)$\\
	$v = \dot{x}_A = x_0 \omega \cos (\omega t)$

	\includegraphics[width=0.5\textwidth]{Images/Energiebilanz beim Feder-Masse-Pendel.png}

	$ \Rightarrow x = x_R + x_A$ \hspace{2cm}
	$ T = 2\pi \sqrt{\frac{m}{D}}$ \hspace{2cm}
	$\omega = 2 \pi f = 2 \pi \frac{1}{T} = \sqrt{\frac{D}{m}}$\\
	\begin{alignat}{1}
		E_{pot} &= m \cdot g \cdot h = -m \cdot g \cdot x = - m \cdot g \cdot (x_R + x_A)\\
		E_{kin} &= \frac{1}{2} \cdot m \cdot v^2\\
		E_{span} &= \frac{1}{2} \cdot D \cdot x^2 = \frac{1}{2} \cdot D \cdot (x_R + x_A)^2\\
		&= \frac{1}{2} \cdot D \cdot (x_R^2 + 2 x_R x_A + x_A^2)\\
		&= \frac{1}{2} \cdot D x_R^2 + D \cdot x_R \cdot x_A + \frac{1}{2} \cdot D \cdot x_A^2
	\end{alignat}
	Für die Ruhelage gilt:
	\begin{alignat}{1}
		F_g &= F_D\\
		m \cdot g &= D \cdot x_R \qquad \text{In $E_{pot}$ einsetzen}\\
	\end{alignat}
	\begin{alignat}{1}
		E &= E_{pot} + E_{kin} + E_{span} \\
		&= -D \cdot x_R \cdot (x_R + x_A) + \frac{1}{2} \cdot m \cdot v^2 + \frac{1}{2} \cdot D \cdot x_R^2 + D\cdot x_R \cdot x_A + \frac{1}{2} \cdot D \cdot x_A^2\\
		&= -Dx_R^2 - Dx_R x_A + \frac{1}{2} mv^2 + \frac{1}{2} D x_R^2+ Dx_R  x_A + \frac{1}{2} D x_A^2\\
		&= -\frac{1}{2} D x_R^2 + \frac12 m v^2 + \frac12 D x_A^2
	\end{alignat}
	$-\frac{1}{2} D x_R^2 \text{ = Konstante}$\\
	$\frac12 m v^2 = E_{kin}$\\
	$\frac12 D x_A^2 =$ Spannenergie bezogen auf die Ruhelage\\

	\section{Das Fadenpendel}

	\begin{alignat}{1}
		\alpha \text{ im Bogenmaß}\\
		\Rightarrow b &= \alpha \cdot l\\
		F_G &= m \cdot g\\
		\sin \alpha &= \frac{F_{\perp}}{F_G}\\
		F_{\perp} &= F_G \sin \alpha
	\end{alignat}

	Beschleunigung der Masse:

	\begin{alignat}{1}
		F &= ma\\
		a &= \frac{F}{m} = \frac{F_G \cdot \sin \alpha}{m} = \frac{mg \cdot \sin \alpha}{m} = g \cdot \sin \alpha\\
	\end{alignat}

	\begin{alignat}{1}
		F_a + F_{\perp} &= 0\\
		m \cdot a + F_G \sin \alpha &= 0\\
		m \cdot \ddot{b} + mg\sin \alpha &= 0\\
		m \cdot \ddot{\alpha} \cdot l + m g \sin \alpha &= 0\\
	\end{alignat}

	\begin{alignat}{1}
		a &= \ddot{b}\\
		b &= \alpha \cdot l\\
		\dot{b} &= \dot{\alpha} \cdot l\\
		\ddot{b} &= \ddot{\alpha} \cdot l
	\end{alignat}

	Im Bogenmaß $\alpha \approx \sin \alpha$ für kleine $\alpha$

	\begin{alignat}{1}
		\Rightarrow m l \ddot{\alpha} + m g \alpha &= 0 \qquad \mid : m \\
		l \ddot{\alpha} + g \alpha &= 0\\
	\end{alignat}

	\begin{alignat}{1}
		\text{Lösungsansatz: } \alpha(t) &= \alpha_0 \sin(\omega t + \varphi)\\
		\dot{\alpha}&=\alpha_0\omega \cos(\omega t+\varphi)\\ \ddot{\alpha}&=-\alpha_0\omega^2 \sin(\omega t+\varphi)\\
		-\alpha_0\omega^2 \sin(\omega t+\varphi)l+g\alpha_0\sin(\omega t+\varphi)&=0\\
		-\omega^2 l+g&=0\\
		\omega^2&=\frac gl\\
		\omega&=\sqrt{\frac gl} \\
		-\alpha_0\omega^2 \sin(\omega t+\pi)l+g\alpha_0\sin(\omega t+\pi)&=0
	\end{alignat}

	\begin{alignat}{1}
		T &= \frac{1}{f} = \frac{2 \pi}{\omega}\\
		\Rightarrow T &= 2 \pi \sqrt{\tfrac{l}{g}}\\
	\end{alignat}
	Für Startbedingungen
	\begin{alignat}{1}
		\alpha (t = 0) &= \alpha_0\\
		\alpha_0 \cdot \sin(\omega \cdot 0 + \varphi) &= \alpha_0\\
		\sin (\varphi) &= 1\\
		\Rightarrow \varphi &= \frac{\pi}{2}\\
	\end{alignat}
	\begin{alignat}{1}
		\Rightarrow \alpha (t) &= \alpha_0 \sin\!\left(\omega t + \frac{\pi}{2}\right)\\
		&= \alpha_0 \cos\!\left(\sqrt{\frac{g}{l}} \cdot t\right)\\
	\end{alignat}

	\noindent
	\textbf{Maschenregel für den Schwingkreis:}\\
	\begin{minipage}[t]{0.3\textwidth}
		\begin{align*}
			U_C + U_L &= 0\\
			\frac{Q}{C} + L \dot{I} &= 0\\
			\frac{Q}{C} + L \ddot{Q} &= 0
		\end{align*}
	\end{minipage}
	\hfill
	\begin{minipage}[t]{0.7\textwidth}
		\begin{align*}
			\textbf{Ansatz: } &Q(t) = \hat{Q} \sin (\omega t + \varphi)\\
				       &\dot{Q}(t) = \hat{Q} \,\omega \cos(\omega t + \varphi)\\
				       &\ddot{Q}(t) = -\hat{Q} \,\omega^2 \sin(\omega t + \varphi)\\
		\end{align*}
	\end{minipage}
	
	\vspace{0.5cm}
	\noindent
	\textbf{Differentialgleichung:}
	\begin{align*}
		\frac{\hat{Q} \sin(\omega t + \varphi)}{C} + L \left(-\hat{Q}\,\omega^2 \sin(\omega t + \varphi)\right) &= 0\\
		\hat{Q} \sin(\omega t + \varphi) \left( \frac{1}{C} - L \omega^2 \right) &= 0\\
		\frac{1}{C} - L \omega^2 &= 0\\
		\omega^2 &= \frac{1}{LC}\\
		\omega &= \frac{1}{\sqrt{LC}}
	\end{align*}
	\begin{align*}
		2 \pi f &= \frac{1}{\sqrt{L C}}\\
		f &= \frac{1}{2 \pi \sqrt{L C}} \text{: Thomsonsche Schwingungsgleichung}
	\end{align*}
